package ty;

import javax.jws.WebMethod;
import javax.jws.WebParam;
import javax.jws.WebService;

@WebService(serviceName = "temperature")
public class temperature {

    @WebMethod(operationName = "FtoC")
    public String FtoC(@WebParam(name = "f") float f) {
        return "In Celsius " + ((f - 32) * 5 / 9);
    }

    @WebMethod(operationName = "CtoF")
    public String CtoF(@WebParam(name = "c") float c) {
        return "In Fahrenheit " + ((c * 9 / 5) + 32);
    }
}
