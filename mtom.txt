package tycs;

import java.io.File;
import java.nio.file.Files;
import javax.jws.Oneway;
import javax.jws.WebMethod;
import javax.jws.WebParam;
import javax.jws.WebService;
import javax.xml.ws.soap.MTOM;

@MTOM(enabled = true, threshold = 6000)
@WebService(serviceName = "imagews1")
public class imagews1 {

    private static final String DIR = "C:\\Users\\User\\Downloads\\server\\";

    @WebMethod(operationName = "hello")
    public String hello(@WebParam(name = "name") String txt) {
        return "Hello " + txt + " !";
    }

    @WebMethod(operationName = "upload")
    @Oneway
    public void upload(@WebParam(name = "Filename") String Filename,
                       @WebParam(name = "ImageBytes") byte[] ImageBytes) {
        try {
            new File(DIR).mkdirs();
            Files.write(new File(DIR + Filename).toPath(), ImageBytes);
            System.out.println("Received file: " + DIR + Filename);
        } catch (Exception ex) {
            System.out.println(ex);
        }
    }

    @WebMethod(operationName = "download")
    public byte[] download(@WebParam(name = "Filename") String Filename) {
        try {
            return Files.readAllBytes(new File(DIR + Filename).toPath());
        } catch (Exception ex) {
            System.out.println(ex);
        }
        return null;
    }
}
